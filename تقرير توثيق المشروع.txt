TypeScript
// lib/admin/adminSessionStore.ts
const activeSessions = new Set<string>();  // ❌ In-Memory (يُفقد عند Restart)

export function isSessionValid(token: string) {
  return activeSessions.has(token);
}
التأثير:

❌ الجلسات تُفقد عند إعادة تشغيل الخادم
